```53:61:zhi2-mobile-shimin/src/pages_transaction/buildingAlbum/index.vue
    enlargePicture(arr, index) {
      const urls = arr.map((url) => resolveMediaUrl(url)).filter(Boolean)
      if (!urls.length) return
      uni.previewImage({
        loop: true,
        urls,
        current: index,
      })
    },
```
