import patterns from '@/patterns/index.js'
import { currencyFormatter } from '@/formatters/index.js'
        ...
    displayValue(item) {
      const v = item && item.value
      if (v === undefined || v === null || v === '') return ''
      const str = String(v)
      if (!patterns.number.test(str)) return v
      return currencyFormatter(v, 0, 20, v)
    }
