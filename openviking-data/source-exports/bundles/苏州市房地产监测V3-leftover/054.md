```6:16:d:\repository\苏州市房地产监测V3\zhi2-web-admin\src\utils\idCard.js
/** 18 位身份证号格式校验（含校验位） */
export function isValidIdNumber(idNumber) {
  if (!idNumber || !ID_CARD_18_REG.test(idNumber)) {
    return false
  }
  const upper = idNumber.toUpperCase()
  let sum = 0
  for (let i = 0; i < 17; i++) {
    sum += parseInt(upper[i], 10) * WEIGHTS[i]
  }
  return CHECK_CODES[sum % 11] === upper[17]
}
```
