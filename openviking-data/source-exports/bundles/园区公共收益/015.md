```101:112:zhi2-web-admin/src/utils/todoTask.js
export function getTodoTaskRoutePath(row) {
  const path = row?.path || ''
  const detailId = row?.detailId != null && row?.detailId !== '' ? String(row.detailId) : ''

  if (detailId && path.endsWith(detailId)) {
    return path
  }
  if (path.startsWith('/')) {
    return path
  }
  return getRouterPathByFilePath(path) + detailId
}
```

| 判断条件 | 处理方式 |
